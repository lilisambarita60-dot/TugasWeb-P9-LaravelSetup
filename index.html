✔ 7. README (di bawah)
✔ 8. Repo: TugasWeb-P9-LaravelSetup
⭐ Bonus: route parameter /hello/{nama}

💻 Kode Laravel

routes/web.php
use Illuminate\Support\Facades\Route;
use App\Http\Controllers\PageController;

Route::get('/', function () {
    return view('welcome', [
        'name' => 'Mahasiswa',
        'courses' => ['HTML', 'CSS', 'Laravel'],
    ]);
});
Route::get('/about', [PageController::class, 'about']);
Route::get('/contact', [PageController::class, 'contact']);
Route::get('/hello/{nama}', function ($nama) {
    return view('hello', ['nama' => $nama]);
});
app/Http/Controllers/PageController.php
class PageController extends Controller
{
    public function about()
    {
        return view('about', ['campus' => 'UNIMED']);
    }
    public function contact()
    {
        return view('contact', ['email' => 'kontak@contoh.test']);
    }
}
resources/views/welcome.blade.php
<h1>Halo, {{ $name }}! 👋</h1>
<ul>
  @foreach ($courses as $c)
    <li>{{ $c }}</li>
  @endforeach
</ul>

📄 README.md

Langkah install 
# 1. Prasyarat
php -v          # PHP >= 8.2
composer -V
node -v

# 2. Buat project
composer create-project laravel/laravel blog
cd blog

# 3. Konfigurasi .env (MySQL)
DB_CONNECTION=mysql
DB_HOST=127.0.0.1
DB_PORT=3306
DB_DATABASE=blog_db
DB_USERNAME=root
DB_PASSWORD=

# 4. Generate file
php artisan make:controller PageController
php artisan make:model Post -m

# 5. Jalankan
php artisan serve   # http://127.0.0.1:8000
Struktur folder
app/ = Model & Controller (M dan C) · resources/views = View (V) · routes/web.php = peta URL · database/ = migrations, factories, seeders · public/ = document root · config/ = konfigurasi · storage/ = upload, cache, log · .env = rahasia lokal (jangan di-commit)
